67 void initScriptAnalysisAndIsolatePairs(Vector<IsolatePair> &isolatePairs)
68 {
69 Vector<int> isolateStack;
70 // load directions of string, and determine isolate pairs
71 for (int i = 0; i < length; ++i) {
72 int pos = i;
73 char32_t uc = text[i].unicode();
74 if (QChar::isHighSurrogate(uc) && i < length - 1 && text[i + 1].isLowSurrogate()) {
75 ++i;
76 analysis[i].bidiDirection = QChar::DirNSM;
77 uc = QChar::surrogateToUcs4(ushort(uc), text[i].unicode());
78 }
79 const auto direction = QChar::direction(uc);
80 analysis[pos].bidiDirection = direction;
81 switch (direction) {
82 case QChar::DirON:
83 // all mirrored chars are DirON
84 if (QChar::mirroredChar(uc) != uc)
85 analysis[pos].bidiFlags = BidiAnalysis::BidiMirrored;
86 break;
87 case QChar::DirLRE:
88 case QChar::DirRLE:
89 case QChar::DirLRO:
90 case QChar::DirRLO:
91 case QChar::DirPDF:
92 case QChar::DirBN:
93 analysis[pos].bidiFlags = BidiAnalysis::BidiMaybeResetToParagraphLevel|BidiAnalysis::BidiBN;
94 break;
95 case QChar::DirLRI:
96 case QChar::DirRLI:
97 case QChar::DirFSI:
98 isolateStack.append(isolatePairs.size());
99 isolatePairs.append({ pos, length });
100 analysis[pos].bidiFlags = BidiAnalysis::BidiMaybeResetToParagraphLevel;
101 break;
102 case QChar::DirPDI:
103 if (!isolateStack.isEmpty()) {
104 isolatePairs[isolateStack.last()].end = pos;
105 isolateStack.removeLast();
106 }
107 Q_FALLTHROUGH()[[clang::fallthrough]];
108 case QChar::DirWS:
109 analysis[pos].bidiFlags = BidiAnalysis::BidiMaybeResetToParagraphLevel;
110 break;
111 case QChar::DirS:
112 case QChar::DirB:
113 analysis[pos].bidiFlags = BidiAnalysis::BidiResetToParagraphLevel;
114 if (uc == QChar::ParagraphSeparator) {
115 // close all open isolates as we start a new paragraph
116 while (!isolateStack.isEmpty()) {
117 isolatePairs[isolateStack.last()].end = pos;
118 isolateStack.removeLast();
119 }
120 }
121 break;
122 default:
123 break;
124 }
125 if (i != pos) {
126 analysis[i].bidiFlags = analysis[pos].bidiFlags;
127 if (analysis[pos].bidiFlags & BidiAnalysis::BidiBN)
128 analysis[i].bidiDirection = QChar::DirBN;
129 }

641 // opening bracked
642 uint closingBracked = canonicalBracket(mirrored);
643 bracketStack.push(closingBracked, bracketPairs.size());
644 if (bracketStack.overflowed()) {
645 bracketPairs.clear();
646 break;
647 }
648 bracketPairs.append({ pos, -1 });
649 } else if (category == QChar::Punctuation_Close
650 && QChar::category(mirrored) == QChar::Punctuation_Open) {
651 int pairPos = bracketStack.match(canonicalBracket(scalar));
652 if (pairPos != -1)
653 bracketPairs[pairPos].second = pos;
654 }
655 }
656 }
657 ++it;
658 }
659 }
660
661 if (BidiDebugEnabled && bracketPairs.size()) {
662 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << "matched bracket pairs:";
663 for (int i = 0; i < bracketPairs.size(); ++i)
664 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << " " << bracketPairs.at(i).first << bracketPairs.at(i).second;
665 }
666
667 QChar::Direction lastStrong = sos;
668 IsolatedRunSequenceIterator it(runs, i);
669 QChar::Direction embeddingDir = (level & 1) ? QChar::DirR : QChar::DirL;
670 IsolatedRunSequenceIterator opening(runs, runIndex);
671 for (int i = 0; i < bracketPairs.size(); ++i) {
672 const auto &pair = bracketPairs.at(i);
673 if (!pair.isValid())
674 continue;
675 // N0 considers only this isolating run sequence, not embedded/isolate contents.
676 while (!opening.atEnd() && *opening != pair.first) ++opening;
677 QChar::Direction containedDir = pair.containedDirection(opening, analysis, embeddingDir);
678 if (containedDir == QChar::DirON) {
679 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << " 3: resolve bracket pair" << i << "to DirON";
680 continue;
681 } else if (containedDir == embeddingDir) {
682 analysis[pair.first].bidiDirection = embeddingDir;
683 analysis[pair.second].bidiDirection = embeddingDir;
684 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << " 1: resolve bracket pair" << i << "to" << embeddingDir;
685 } else {
686 // case c.
687 while (it.pos < pair.first) {
688 int pos = *it;
689 switch (analysis[pos].bidiDirection) {
690 case QChar::DirR:
691 case QChar::DirEN:
692 case QChar::DirAN:
693 lastStrong = QChar::DirR;
694 break;
695 case QChar::DirL:
696 lastStrong = QChar::DirL;
697 break;
698 default:
699 break;
700 }
701 ++it;
702 }
703 analysis[pair.first].bidiDirection = lastStrong;
704 analysis[pair.second].bidiDirection = lastStrong;
705 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << " 2: resolve bracket pair" << i << "to" << lastStrong;
706 }
707 // N0 applies to original NSMs after either bracket, ignoring X9 removals.
708 for (int bracket : {pair.first, pair.second}) {
709 auto trailing = opening;
710 while (!trailing.atEnd() && *trailing != bracket) ++trailing;
711 if (trailing.atEnd()) continue;
712 ++trailing;
713 while (!trailing.atEnd()) {
714 const int pos = *trailing;
715 if (analysis[pos].bidiFlags & BidiAnalysis::BidiBN) {
716 ++trailing;
717 continue;
718 }
719 if (originalDirection(pos) != QChar::DirNSM) break;
720 analysis[pos].bidiDirection = analysis[bracket].bidiDirection;
721 ++trailing;
722 }
723 }
724 }
725 }

727 void resolveN1N2(const Vector<DirectionalRun> &runs, int i, QChar::Direction sos, QChar::Direction eos)
728 {
729 // Rule N1 & N2
730 QChar::Direction lastStrong = sos;
731 IsolatedRunSequenceIterator::Position niPos;
732 IsolatedRunSequenceIterator it(runs, i);
733// QChar::Direction last = QChar::DirON;
734 while (1) {
735 int pos = *it;
736
737 QChar::Direction current = pos >= 0 ? analysis[pos].bidiDirection : eos;
738 QChar::Direction currentStrong = current;
739 switch (current) {
740 case QChar::DirEN:
741 case QChar::DirAN:
742 currentStrong = QChar::DirR;
743 Q_FALLTHROUGH()[[clang::fallthrough]];
744 case QChar::DirL:
745 case QChar::DirR:
746 if (niPos.isValid()) {
747 QChar::Direction dir = currentStrong;
748 if (lastStrong != currentStrong)
749 dir = (runs.at(i).level) & 1 ? QChar::DirR : QChar::DirL;
750 it.setPosition(niPos);
751 while (*it != pos) {
752 if (analysis[*it].bidiDirection != QChar::DirBN)
753 analysis[*it].bidiDirection = dir;
754 ++it;
755 }
756 niPos.clear();
757 }
758 lastStrong = currentStrong;
759 break;
760
761 case QChar::DirBN:
762 case QChar::DirS:
763 case QChar::DirWS:
764 case QChar::DirON:
765 case QChar::DirFSI:
766 case QChar::DirLRI:
767 case QChar::DirRLI:
768 case QChar::DirPDI:
769 case QChar::DirB:
770 if (!niPos.isValid())
771 niPos = it.position();
772 break;
773
774 default:
775 Q_UNREACHABLE()do { static_cast<void>(false && (false)); __builtin_unreachable
(); } while (false);
776 }
777 if (it.atEnd())
778 break;
779// last = current;
780 ++it;
781 }

784 void resolveImplicitLevelsForIsolatedRun(const Vector<DirectionalRun> &runs, int i)
785 {
786 // Rule X10
787 int level = runs.at(i).level;
788 int before = i - 1;
789 while (before >= 0 && !runs.at(before).hasContent)
790 --before;
791 int level_before = (before >= 0) ? runs.at(before).level : baseLevel;
792 int after = i;
793 while (runs.at(after).continuation >= 0)
794 after = runs.at(after).continuation;
795 if (runs.at(after).continuation == -2) {
796 after = runs.size();
797 } else {
798 ++after;
799 while (after < runs.size() && !runs.at(after).hasContent)
800 ++after;
801 }
802 int level_after = (after == runs.size()) ? baseLevel : runs.at(after).level;
803 QChar::Direction sos = (qMax(level_before, level) & 1) ? QChar::DirR : QChar::DirL;
804 QChar::Direction eos = (qMax(level_after, level) & 1) ? QChar::DirR : QChar::DirL;
805
806 if (BidiDebugEnabled) {
807 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << "Isolated run starting at" << i << "sos/eos" << sos << eos;
808 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << "before implicit level processing:";
809 IsolatedRunSequenceIterator it(runs, i);
810 while (!it.atEnd()) {
811 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << " " << *it << Qt::hex << text[*it].unicode() << analysis[*it].bidiDirection;
812 ++it;
813 }
814 }
815
816 resolveW1W2W3(runs, i, sos);
817 resolveW4(runs, i, sos);
818 resolveW5(runs, i);
819
820 if (BidiDebugEnabled) {
821 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << "after W4/W5";
822 IsolatedRunSequenceIterator it(runs, i);
823 while (!it.atEnd()) {
824 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << " " << *it << Qt::hex << text[*it].unicode() << analysis[*it].bidiDirection;
825 ++it;
826 }
827 }
828
829 resolveW6W7(runs, i, sos);
830
831 // Resolve neutral types
832
833 // Rule N0
834 resolveN0(runs, i, sos);
835 resolveN1N2(runs, i, sos, eos);
836
837 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << "setting levels (run at" << level << ")";
838 // Rules I1 & I2: set correct levels
839 {
840 ushort level = runs.at(i).level;
841 IsolatedRunSequenceIterator it(runs, i);
842 while (!it.atEnd()) {
843 int pos = *it;
844
845 QChar::Direction current = analysis[pos].bidiDirection;
846 switch (current) {
847 case QChar::DirBN:
848 break;
849 case QChar::DirL:
850 analysis[pos].bidiLevel = (level + 1) & ~1;
851 break;
852 case QChar::DirR:
853 analysis[pos].bidiLevel = level | 1;
854 break;
855 case QChar::DirAN:
856 case QChar::DirEN:
857 analysis[pos].bidiLevel = (level + 2) & ~1;
858 break;
859 default:
860 Q_UNREACHABLE()do { static_cast<void>(false && (false)); __builtin_unreachable
(); } while (false);
861 }
862 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << " " << pos << current << analysis[pos].bidiLevel;
863 ++it;
864 }
865 }
866 }
867
868 void resolveImplicitLevels(const Vector<DirectionalRun> &runs)
869 {
870 for (int i = 0; i < runs.size(); ++i) {
871 if (runs.at(i).isContinuation)
872 continue;
873
874 resolveImplicitLevelsForIsolatedRun(runs, i);
875 }
876 }

878 bool checkForBidi() const
879 {
880 if (baseLevel != 0)
881 return true;
882 for (int i = 0; i < length; ++i) {
883 if (text[i].unicode() >= 0x590) {
884 switch (originalDirection(i)) {
885 case QChar::DirR: case QChar::DirAN:
886 case QChar::DirLRE: case QChar::DirLRO: case QChar::DirAL:
887 case QChar::DirRLE: case QChar::DirRLO: case QChar::DirPDF:
888 case QChar::DirLRI: case QChar::DirRLI: case QChar::DirFSI: case QChar::DirPDI:
889 return true;
890 default:
891 break;
892 }
893 }
894 }
895 return false;
896 }
897
898 QChar::Direction originalDirection(int position) const {
899 const QChar ch = text[position];
900 if (ch.isLowSurrogate() && position > 0 && text[position - 1].isHighSurrogate())
901 return QChar::DirNSM;
902 char32_t scalar = ch.unicode();
903 if (ch.isHighSurrogate() && position + 1 < length && text[position + 1].isLowSurrogate())
904 scalar = QChar::surrogateToUcs4(ch, text[position + 1]);
905 return QChar::direction(scalar);
906 }
907
908 static uint canonicalBracket(char32_t scalar) {
909 const auto normalized = QString::fromUcs4(&scalar, 1)
910 .normalized(QString::NormalizationForm_D).toUcs4();
911 Q_ASSERT(normalized.size() == 1)static_cast<void>(false && (normalized.size() ==
1));
912 return normalized.front();
913 }
914
915 bool process()
916 {
917 memset(analysis, 0, length * sizeof(BidiAnalysis));
918
919 bool hasBidi = checkForBidi();
920
921 if (!hasBidi)
922 return false;
923
924 if (BidiDebugEnabled) {
925 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << ">>>> start bidi, text length" << length;
926 for (int i = 0; i < length; ++i)
927 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << Qt::hex << " (" << i << ")" << text[i].unicode() << text[i].direction();
928 }
929
930 {
931 Vector<DirectionalRun> runs;
932 resolveExplicitLevels(runs);
933
934 if (BidiDebugEnabled) {
935 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << "resolved explicit levels, nruns" << runs.size();
936 for (int i = 0; i < runs.size(); ++i)
937 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << " " << i << "start/end" << runs.at(i).start << runs.at(i).end << "level" << (int)runs.at(i).level << "continuation" << runs.at(i).continuation;
938 }
939
940 // now we have a list of isolated run sequences inside the vector of runs, that can be fed
941 // through the implicit level resolving
942
943 resolveImplicitLevels(runs);
944 }
945
946 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << "Rule L1:";
947 // Rule L1:
948 bool resetLevel = true;
949 for (int i = length - 1; i >= 0; --i) {
950 if (analysis[i].bidiFlags & BidiAnalysis::BidiResetToParagraphLevel) {
951 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << "resetting pos" << i << "to baselevel";
952 analysis[i].bidiLevel = baseLevel;
953 resetLevel = true;
954 } else if (resetLevel && analysis[i].bidiFlags & BidiAnalysis::BidiMaybeResetToParagraphLevel) {
955 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << "resetting pos" << i << "to baselevel (maybereset flag)";
956 analysis[i].bidiLevel = baseLevel;
957 } else {
958 resetLevel = false;
959 }
960 }
961
962 // set directions for BN to the minimum of adjacent chars
963 // This makes is possible to be conformant with the Bidi algorithm even though we don't
964 // remove BN and explicit embedding chars from the stream of characters to reorder
965 int lastLevel = baseLevel;
966 int lastBNPos = -1;
967 for (int i = 0; i < length; ++i) {
968 if (analysis[i].bidiFlags & BidiAnalysis::BidiBN) {
969 if (lastBNPos < 0)
970 lastBNPos = i;
971 analysis[i].bidiLevel = lastLevel;
972 } else {
973 int l = analysis[i].bidiLevel;
974 if (lastBNPos >= 0) {
975 if (l < lastLevel) {
976 while (lastBNPos < i) {
977 analysis[lastBNPos].bidiLevel = l;
978 ++lastBNPos;
979 }
980 }
981 lastBNPos = -1;
982 }
983 lastLevel = l;
984 }
985 }
986 if (lastBNPos >= 0 && baseLevel < lastLevel) {
987 while (lastBNPos < length) {
988 analysis[lastBNPos].bidiLevel = baseLevel;
989 ++lastBNPos;
990 }
991 }
992
993 if (BidiDebugEnabled) {
994 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << "final resolved levels:";
995 for (int i = 0; i < length; ++i)
996 BIDI_DEBUGif (1) ; else QMessageLogger(nullptr, 0, nullptr).debug() << " " << i << Qt::hex << text[i].unicode() << Qt::dec << (int)analysis[i].bidiLevel;
997 }
998
999 return true;
1000 }

1012Paragraph resolve(const QString &text, Qt::LayoutDirection direction) {
1013 if (text.size() > std::numeric_limits<int>::max())
1014 throw std::length_error("Bidi text exceeds index range");
1015 Paragraph result;
1016 result.base = direction == Qt::RightToLeft
1017 || (direction == Qt::LayoutDirectionAuto && text.isRightToLeft());
1018 if (text.isEmpty()) return result;
1019 std::vector<BidiAnalysis> analysis(text.size());
1020 QBidiAlgorithm algorithm(text.constData(), analysis.data(), text.size(), result.base != 0);
1021 algorithm.process();
1022 result.levels.reserve(analysis.size());
1023 for (const auto &item : analysis) result.levels.push_back(item.bidiLevel);
1024 return result;
1025}

1027std::vector<int> reorder(const std::vector<quint8> &levels) {
1028 if (levels.size() > size_t(std::numeric_limits<int>::max()))
1029 throw std::length_error("Bidi levels exceed index range");
1030 std::vector<int> order(levels.size());
1031 std::iota(order.begin(), order.end(), 0);
1032 if (levels.empty()) return order;
1033 const auto bounds = std::minmax_element(levels.begin(), levels.end());
1034 const int lowestOdd = *bounds.first | 1;
1035 for (int level = *bounds.second; level >= lowestOdd; --level) {
1036 for (size_t begin = 0; begin < levels.size();) {
1037 if (levels[begin] < level) { ++begin; continue; }
1038 size_t end = begin + 1;
1039 while (end < levels.size() && levels[end] >= level) ++end;
1040 std::reverse(order.begin() + begin, order.begin() + end);
1041 begin = end;
1042 }
1043 }
1044 return order;
