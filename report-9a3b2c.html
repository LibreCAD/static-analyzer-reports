119// Case-insensitive lookup of a field by name; returns -1 if not present.
120int findFieldCI(DBFHandle dbf, const std::initializer_list<const char*>& names) {
121 if (!dbf) return -1;
122 const int nFields = DBFGetFieldCount(dbf);
123 for (int f = 0; f < nFields; ++f) {
124 char name[16] = {};
125 int w = 0, d = 0;
126 (void)DBFGetFieldInfo(dbf, f, name, &w, &d);
127 for (const char* candidate : names) {
128 if (qstricmp(name, candidate) == 0) return f;
129 }
130 }
131 return -1;
132}

134// Populate options: read RS_Settings overrides first, else auto-detect from
135// the DBF header. Returns the resolved field-index bundle for the caller.
136struct ResolvedFields {
137 int layer{-1};
138 int color{-1};
139 int ltype{-1};
140 int width{-1};
141 int label{-1};
142};

144ResolvedFields resolveFields(DBFHandle dbf, LC_ShpImportOptions& opt) {
145 // Settings-driven overrides (empty = auto-detect).
146 LC_GROUPRS_Settings::instance()->beginGroup("ShpImport");
147 if (opt.layerField.isEmpty())
148 opt.layerField = LC_GET_STRRS_Settings::instance()->readStr("LayerField", "");
149 if (opt.colorField.isEmpty())
150 opt.colorField = LC_GET_STRRS_Settings::instance()->readStr("ColorField", "");
151 if (opt.ltypeField.isEmpty())
152 opt.ltypeField = LC_GET_STRRS_Settings::instance()->readStr("LineTypeField", "");
153 if (opt.widthField.isEmpty())
154 opt.widthField = LC_GET_STRRS_Settings::instance()->readStr("WidthField", "");
155 if (opt.labelField.isEmpty())
156 opt.labelField = LC_GET_STRRS_Settings::instance()->readStr("LabelField", "");
157 // opt.labelHeight left at the default 2.0; RS_Settings has no double
158 // accessor, so plumbing a numeric override into the /ShpImport group is
159 // deferred to a future options dialog (Phase 2c non-goal).
160 LC_GROUP_ENDRS_Settings::instance()->endGroup();
161 opt.importLabels = LC_GET_ONE_BOOLRS_Settings::instance()->readBoolSingle("ShpImport", "ImportLabels", true);
162
163 ResolvedFields r;
164 if (!dbf) return r;
165
166 // Explicit override wins over auto-detect for each field.
167 auto lookupOrDetect = [&](const QString& explicitName,
168 std::initializer_list<const char*> defaults) {
169 if (!explicitName.isEmpty()) {
170 const QByteArray bytes = explicitName.toLatin1();
171 return findFieldCI(dbf, {bytes.constData()});
172 }
173 return findFieldCI(dbf, defaults);
174 };
175
176 r.layer = lookupOrDetect(opt.layerField, {"LAYER", "LEVEL", "LYR"});
177 r.color = lookupOrDetect(opt.colorField, {"COLOR", "COLOUR"});
178 r.ltype = lookupOrDetect(opt.ltypeField, {"LINETYPE", "LTYPE"});
179 r.width = lookupOrDetect(opt.widthField, {"WIDTH", "LWEIGHT", "LINEWT"});
180 r.label = lookupOrDetect(opt.labelField, {"NAME", "LABEL", "TEXT"});
181 return r;
182}

184// ---------------------------------------------------------------------------
185// DBF codepage → QStringDecoder.
186// ---------------------------------------------------------------------------
187
188// Decode a DBF string field with the DBF's declared codepage. Falls back to
189// UTF-8 → Latin-1 on invalid sequences (the same pragmatic heuristic
190// rs_filterjww.cpp uses for JWW's mixed CP932/Shift_JIS content).
191QString decodeDbfString(const char* raw, const QString& codepage) {
192 if (raw == nullptr || *raw == '\0') return {};
193 const QByteArray bytes(raw);
194
195 // Best-effort mapping from DBF codepage marker to a Qt codec name.
196 // shapelib returns strings like "UTF-8", "LDID/87" (Latin-1 / ISO-8859-1),
197 // "LDID/13" (CP850), etc. We only try to hit the common ones; anything
198 // unrecognised falls through to UTF-8 with Latin-1 fallback.
199 auto encodingFromDbf = [](const QString& cp) -> std::optional<QStringConverter::Encoding> {
200 const QString u = cp.toUpper();
201 if (u == "UTF-8" || u == "UTF8" || u == "LDID/8B") // LDID 0x8B = UTF-8 (unofficial)
202 return QStringConverter::Utf8;
203 if (u == "LDID/87" || u == "LDID/57" || u == "ISO-8859-1" || u == "LATIN1"
204 || u == "CP1252" || u == "WINDOWS-1252")
205 return QStringConverter::Latin1;
206 return std::nullopt;
207 };
208
209 if (auto enc = encodingFromDbf(codepage)) {
210 QStringDecoder decoder{*enc};
211 QString result = decoder(bytes);
212 if (!decoder.hasError()) return result;
213 }
214 // UTF-8 first, then Latin-1.
215 {
216 QStringDecoder utf8{QStringConverter::Utf8};
217 QString result = utf8(bytes);
218 if (!utf8.hasError()) return result;
219 }
220 QStringDecoder latin1{QStringConverter::Latin1};
221 return latin1(bytes);
222}

228// Numeric COLOR field: 0-255 → ACI palette (reuse RS_FilterDXFRW::numberToColor),
229// larger → 24-bit RGB (0xRRGGBB).
230RS_Color colorFromNumber(int v) {
231 if (v >= 0 && v <= 255) return RS_FilterDXFRW::numberToColor(v);
232 return RS_Color((v >> 16) & 0xFF, (v >> 8) & 0xFF, v & 0xFF);
233}

235// Nearest-neighbour: pick the RS2::LineWidth enum value closest to the
236// requested numeric weight (interpreted as tenths-of-mm, matching the
237// enum's own integer encoding — see rs.h:867-897).
238RS2::LineWidth widthFromNumber(double v) {
239 const int enumVal = static_cast<int>(std::lround(v));
240 const RS2::LineWidth candidates[] = {
241 RS2::Width00, RS2::Width01, RS2::Width02, RS2::Width03, RS2::Width04,
242 RS2::Width05, RS2::Width06, RS2::Width07, RS2::Width08, RS2::Width09,
243 RS2::Width10, RS2::Width11, RS2::Width12, RS2::Width13, RS2::Width14,
244 RS2::Width15, RS2::Width16, RS2::Width17, RS2::Width18, RS2::Width19,
245 RS2::Width20, RS2::Width21, RS2::Width22, RS2::Width23};
246 RS2::LineWidth best = RS2::Width00;
247 int bestDist = std::abs(static_cast<int>(RS2::Width00) - enumVal);
248 for (RS2::LineWidth w : candidates) {
249 const int d = std::abs(static_cast<int>(w) - enumVal);
250 if (d < bestDist) { bestDist = d; best = w; }
251 }
252 return best;
253}

255// Build an RS_Pen from resolved fields for one DBF record. If no styling
256// fields are present, returns std::nullopt so callers can leave the entity
257// at ByLayer defaults.
258std::optional<RS_Pen> penFromRecord(DBFHandle dbf, int record,
259 const ResolvedFields& rf,
260 const QString& codepage) {
261 if (!dbf || record < 0) return std::nullopt;
262 bool any = false;
263 RS_Pen pen{RS_Color(RS2::FlagByLayer), RS2::WidthByLayer, RS2::LineByLayer};
264
265 if (rf.color >= 0 && !DBFIsAttributeNULL(dbf, record, rf.color)) {
266 const int v = DBFReadIntegerAttribute(dbf, record, rf.color);
267 pen.setColor(colorFromNumber(v));
268 any = true;
269 }
270 if (rf.ltype >= 0 && !DBFIsAttributeNULL(dbf, record, rf.ltype)) {
271 const QString name = decodeDbfString(
272 DBFReadStringAttribute(dbf, record, rf.ltype), codepage);
273 if (!name.isEmpty()) {
274 pen.setLineType(RS_FilterDXFRW::nameToLineType(name));
275 any = true;
276 }
277 }
278 if (rf.width >= 0 && !DBFIsAttributeNULL(dbf, record, rf.width)) {
279 const double w = DBFReadDoubleAttribute(dbf, record, rf.width);
280 pen.setWidth(widthFromNumber(w));
281 any = true;
282 }
283 if (!any) return std::nullopt;
284 return pen;
285}

319// Emit one polyline covering the [start, start+count) vertex slice of @p o.
320// Ring convention: for closed polylines the SHP-required duplicated closing
321// vertex (first==last) is dropped before setClosed(true). Degenerate parts
322// (< 2 vertices open, < 3 closed) are skipped.
323void emitOnePart(RS_Graphic& g, RS_Layer* layer,
324 const std::optional<RS_Pen>& pen,
325 const SHPObject& o, int start, int count, bool closed,
326 int& partsEmitted, int& partsSkipped) {
327 if (start < 0 || count <= 0 || start + count > o.nVertices) {
328 ++partsSkipped;
329 return;
330 }
331 if (closed && count >= 2) {
332 const RS_Vector first(o.padfX[start], o.padfY[start]);
333 const RS_Vector last(o.padfX[start + count - 1],
334 o.padfY[start + count - 1]);
335 if (first.squaredTo(last) < 1e-20) --count;
336 }
337 const int minVerts = closed ? 3 : 2;
338 if (count < minVerts) { ++partsSkipped; return; }
339
340 auto* pl = new RS_Polyline(&g, RS_PolylineData(
341 RS_Vector(false), RS_Vector(false), false));
342 pl->setLayer(layer);
343 if (pen) pl->setPen(*pen);
344 for (int i = 0; i < count; ++i) {
345 const int idx = start + i;
346 const double z = (o.padfZ != nullptr) ? o.padfZ[idx] : 0.0;
347 pl->addVertex(RS_Vector(o.padfX[idx], o.padfY[idx], z));
348 }
349 if (closed) pl->setClosed(true);
350 pl->endPolyline();
351 g.addEntity(pl);
352 ++partsEmitted;
