529/**
530 * Updates the Inserts (letters) of this text. Called when the
531 * text or it's m_data, position, alignment, .. changes.
532 * This method also updates the usedTextWidth / usedTextHeight property.
533 */
534void RS_MText::update() {
535 RS_DEBUGRS_Debug::instance()->print("RS_MText::update");
536
537 clear();
538 if (isDeleted()) {
539 return;
540 }
541
542 m_usedTextWidth = 0.0;
543 m_usedTextHeight = 0.0;
544
545 RS_Font* font{RS_FONTLISTRS_FontList::instance()->requestFont(m_data.style)};
546 if (nullptr == font) {
547 return;
548 }
549
550 // letterSpace and space are always treated as positive advances here; the
551 // bidi pass already produced visual L→R order, so the cursor unconditionally
552 // moves rightward during emission.RS_Vector letterPos{0.0, -9.0};
553 RS_Vector letterPos{0.0, -9.0};
554 const RS_Vector letterSpace{font->getLetterSpacing(), 0.0};
555 const double spaceWidth = font->getWordSpacing();
556
557 int lineCounter{0};
558
559 // Every single text line gets stored in this entity container
560 // so we can move the whole line around easily:
561 auto oneLine{new LC_TextLine(this)};
562
563 // First every text line is created with
564 // alignment: top left
565 // angle: 0
566 // height: 9.0
567 // Rotation, scaling and centering is done later
568
569 // Logical-order pass: process MText format codes and accumulate per-line
570 // segments. Each line's segments are then bidi-reordered and emitted via
571 // flushBidiLine(); see the comment on LC_BidiSegment in the header.
572 std::vector<LC_BidiSegment> segments;
573
574 auto closeLine = [&]() {
575 flushBidiLine(*oneLine, segments, letterSpace, letterPos);
576 updateAddLine(oneLine, lineCounter++);
577 oneLine = new LC_TextLine(this);
578 letterPos = RS_Vector(0.0, -9.0);
579 };
580 for (decltype(m_data.text.length()) i = 0; i < m_data.text.length(); ++i) {
581 auto remaining = m_data.text.mid(i);
582 // Handle \F not followed by {<codePage>}
583 if (remaining.startsWith(R"(\F)") && remaining.indexOf(R"(^\\[Ff]\{[\d\w]*\})") != 0) {
584 segments.push_back(
585 {LC_BidiSegment::Char, m_data.text.at(i), font, 0.0, {}, {}});
586 continue;
587 }
588 if (remaining.startsWith(R"(\\)")) {
589 // Allow escape '\', needed to support "\S" and "\P" in string
590 // "\S" is used for super/subscripts
591 // "\P" is used to start a new line
592 // "\\S" and "\\P" to get literal strings "\S" and "\P"
593 segments.push_back(
594 {LC_BidiSegment::Char, m_data.text.at(i++), font, 0.0, {}, {}});
595 continue;
596 }
597
598 switch (m_data.text.at(i).unicode()) {
599 case 0x0A:
600 // line feed:
601 closeLine();
602 break;
603
604 case 0x20:
605 // Space:
606 segments.push_back(
607 {LC_BidiSegment::Space, QChar(' '), nullptr, spaceWidth, {}, {}});
608 break;
609
610 case 0x5C: {
611 bool handled{false};
612 // code (e.g. \S, \P, ..)
613 ++i;
614 if (static_cast<int>(m_data.text.length()) <= i) {
615 continue;
616 }
617 // `i` is a UTF-16 code-unit index into data.text; index the QString
618 // directly. The previous `data.text.toUcs4().at(i)` indexed a UCS-4
619 // array with a UTF-16 offset — out of bounds / wrong codepoint once any
620 // non-BMP (surrogate-pair) char precedes `i`. All inner cases below are
621 // ASCII ('P','f','F','S'), so a BMP code unit is sufficient and a
622 // non-BMP unit correctly falls to default (--i).
623 char16_t ch{m_data.text.at(i).unicode()};
624 switch (ch) {
625 case 'P':
626 closeLine();
627 handled = true;
628 break;
629
630 case 'f':
631 case 'F': {
632 // font change
633 // \f{symbol} changes font to symbol
634 // \f{} sets font to standard
635 ++i;
636 if ('{' != m_data.text.at(i).unicode()) {
637 --i;
638 continue;
639 }
640
641 const qsizetype j{m_data.text.indexOf('}', i)};
642 if (j > i) {
643 QString fontName = m_data.text.mid(i + 1, j - i - 1);
644
645 const auto requestedFont = RS_FONTLISTRS_FontList::instance()->requestFont(fontName);
646 if (requestedFont != nullptr) {
647 font = requestedFont;
648 }
649 else {
650 font = RS_FONTLISTRS_FontList::instance()->requestFont("standard");
651 }
652 i = j;
653 }
654 continue;
655 } // inner case 'f','F'
656
657 case 'S': {
658 QString upperText;
659 QString lowerText;
660
661 // get upper string:
662 ++i; // fixme - sand - rework cycles
663 while (static_cast<int>(m_data.text.length()) > i && m_data.text.at(i).unicode() != '^' && m_data.text.at(i).unicode() !=
664 '\\') {
665 upperText += m_data.text.at(i);
666 ++i;
667 }
668
669 ++i;
670 if (static_cast<int>(m_data.text.length()) > i && '^' == m_data.text.at(i - 1).unicode() && ' ' == m_data.text.at(i).
671 unicode()) {
672 ++i;
673 }
674
675 // get lower string:
676 while (static_cast<int>(m_data.text.length()) > i && ';' != m_data.text.at(i).unicode()) {
677 lowerText += m_data.text.at(i);
678 ++i;
679 }
680
681 LC_BidiSegment stack;
682 stack.kind = LC_BidiSegment::Stack;
683 stack.upperText = std::move(upperText);
684 stack.lowerText = std::move(lowerText);
685 segments.push_back(std::move(stack));
686 handled = true;
687
688 break;
689 } // inner case 'S'
690
691 default:
692 --i;
693 break;
694 } // inner switch (ch)
695
696 if (handled) {
697 break;
698 }
699 } // outer case 0x5C
700
701 // if char is not handled
702 // fall-through
703 default: {
704 // One Letter:
705 segments.push_back(
706 {LC_BidiSegment::Char, m_data.text.at(i), font, 0.0, {}, {}});
707 break;
708 } // outer default
709 } // outer switch (m_data.text.at(i).unicode())
710 } // for (i) loop
711
712 // Flush any trailing segments belonging to the last line.
713 flushBidiLine(*oneLine, segments, letterSpace, letterPos);
714
715 m_usedTextHeight -=
716 m_data.height * m_data.lineSpacingFactor * 5.0 / 3.0 - m_data.height;
717
718 updateAddLine(oneLine, lineCounter);
719
720 alignVertically();
721 RS_DEBUGRS_Debug::instance()->print("RS_MText::update: OK");
722}
723
724/**
725 * Reorder @p segments via the Unicode Bidirectional Algorithm and emit each
726 * one (letter / space / super-sub stack) into @p oneLine in visual L→R order.
727 * Cursor advances are computed from @c .lff metrics, not from QTextLayout's
728 * Qt-font metrics — QTextLayout is used only for the logical→visual
729 * permutation. Clears @p segments on return.
730 */
731void RS_MText::flushBidiLine(LC_TextLine &oneLine,
732 std::vector<LC_BidiSegment> &segments,
733 const RS_Vector &letterSpace,
734 RS_Vector &letterPosition) {
735 if (segments.empty()) {
736 return;
737 }
738
739 // Build a plain-text string for the bidi pass. Stacks become U+FFFC (Object
740 // Replacement Character), which has bidi class ON (Other Neutral) so it
741 // gets resolved to the surrounding context like punctuation.
742 QString plainText;
743 plainText.reserve(static_cast<int>(segments.size()));
744 for (const auto &seg : segments) {
745 switch (seg.kind) {
746 case LC_BidiSegment::Char:
747 plainText.append(seg.codepoint);
748 break;
749 case LC_BidiSegment::Space:
750 plainText.append(QChar(' '));
751 break;
752 case LC_BidiSegment::Stack:
753 plainText.append(QChar(0xFFFC));
754 break;
755 }
756 }
757
758 // For an explicit RightToLeft setting, lay out characters in pure
759 // positional reverse — matching AutoCAD's `drawingDirection` semantics and
760 // the property-panel/dialog widget mirrors. UAX#9 leaves EN digits
761 // direction-immune, so a UAX#9 pass would render "1234" identically in LTR
762 // and RTL on the canvas while the widget shows "4321" — diverging from the
763 // user's edited buffer. The other settings (LeftToRight, TopToBottom,
764 // ByStyle) keep going through UAX#9 with an LTR base so embedded
765 // strong-RTL runs (Hebrew/Arabic) still display correctly.
766 //
767 // Surrogate-pair caveat: segments are built one QChar each, so a non-BMP
768 // codepoint already arrives as two adjacent segments. Reversal swaps the
769 // halves; rendering then fails the font lookup on each half and falls
770 // back to U+FFFD anyway (addLetter is QChar-scoped). Fixing this needs
771 // grapheme-aware segment building plus codepoint-based font lookup —
772 // broader rendering work, not bidi.
773 std::vector<int> visualOrder;
774 if (m_data.drawingDirection == RS_MTextData::RightToLeft) {
775 visualOrder.resize(segments.size());
776 for (size_t i = 0; i < segments.size(); ++i) {
777 visualOrder[i] = static_cast<int>(segments.size() - 1 - i);
778 }
779 } else {
780 visualOrder = computeBidiVisualOrder(plainText, Qt::LeftToRight);
781 }
782
783 for (int logIdx : visualOrder) {
784 const auto &seg = segments[logIdx];
785 switch (seg.kind) {
786 case LC_BidiSegment::Char: {
787 RS_Font *segFont = seg.font;
788 if (segFont == nullptr) {
789 // Fallback: should not normally happen, but stay defensive.
790 segFont = RS_FONTLISTRS_FontList::instance()->requestFont(m_data.style);
791 }
792 if (segFont != nullptr) {
793 addLetter(oneLine, seg.codepoint, *segFont, letterSpace,
794 letterPosition);
795 }
796 break;
797 }
798 case LC_BidiSegment::Space:
799 letterPosition.x += seg.width;
800 break;
801 case LC_BidiSegment::Stack: {
802 double upperWidth = 0.0;
803 if (!seg.upperText.isEmpty()) {
804 RS_MText *upper = createUpperLower(seg.upperText, m_data,
805 letterPosition + RS_Vector{0., 9.});
806 oneLine.addEntity(upper);
807 upper->reparent(&oneLine);
808 upperWidth = upper->getSize().x;
809 }
810 double lowerWidth = 0.0;
811 if (!seg.lowerText.isEmpty()) {
812 RS_MText *lower = createUpperLower(seg.lowerText, m_data,
813 letterPosition + RS_Vector{0., 4.});
814 oneLine.addEntity(lower);
815 lower->reparent(&oneLine);
816 lowerWidth = lower->getSize().x;
817 }
818 letterPosition.x += std::max(upperWidth, lowerWidth);
819 letterPosition += letterSpace;
820 break;
821 }
822 }
823 }
824
825 segments.clear();
826}

888 * @param letter

901 auto letterText{QString(letter)};
902 if (nullptr == font.findLetter(letterText)) {
903 RS_DEBUGRS_Debug::instance()->print("RS_MText::update: missing font for letter( %s ), replaced " "it with QChar(0xfffd)", qPrintable(letterText)QtPrivate::asString(letterText).toLocal8Bit().constData());
904 letterText = QChar(0xfffd);

906
907 LC_LOGRS_Debug::Log() << "RS_MText::update: insert a letter at pos:(" << letterPosition.x << ", " << letterPosition.y << ")";
908
909 const RS_InsertData d(letterText, letterPosition, RS_Vector(1.0, 1.0), 0.0, 1, 1,
910 RS_Vector(0.0, 0.0), font.getLetterList(), RS2::NoUpdate);
911
912 const auto letterEntity{new RS_Insert(this, d)};
913 letterEntity->setPen(RS_Pen(RS2::FlagInvalid));
914 letterEntity->setLayer(nullptr);
915 letterEntity->update();
916 letterEntity->forcedCalculateBorders();
917
918 // Add spacing, if the font is actually wider than word spacing
919 double actualWidth = letterEntity->getMax().x - letterEntity->getMin().x;
920 if (actualWidth >= font.getWordSpacing() + RS_TOLERANCE1.0e-10) {
921 const double letterSpacing = std::max(1., letterSpace.x);
922 const double wordSpacing = font.getWordSpacing();
923 actualWidth = wordSpacing + letterSpacing - std::fmod(actualWidth - wordSpacing, letterSpacing);
924 }
925 LC_LOGRS_Debug::Log() << __LINE__925 << ": actualWidth: " << actualWidth;
926
927 letterPosition.x += actualWidth;
928
929 oneLine.addEntity(letterEntity);
930
931 // next letter position:
932 letterPosition += letterSpace;
933}

956/**
957 * Used internally by update() to add a text line created with
958 * default values and alignment to this text container.
959 *
960 * @param textLine The text line.
961 * @param lineCounter Line number.
962 *
963 * @return distance over the text base-line
964 */
965double RS_MText::updateAddLine(LC_TextLine* textLine, const int lineCounter) {
966 constexpr double ls = 5.0 / 3.0;
967
968 RS_DEBUGRS_Debug::instance()->print("RS_MText::updateAddLine: width: %f", textLine->getSize().x);
969
970 // textLine->forcedCalculateBorders();
971 // RS_DEBUG->print("RS_MText::updateAddLine: width 2: %f",
972 // textLine->getSize().x);
973
974 // Scale:
975 const double scale = m_data.height / 9.0;
976 textLine->scale(RS_Vector{0., 0.}, RS_Vector(scale, scale));
977
978 textLine->forcedCalculateBorders();
979
980 // Horizontal Align:
981 switch (m_data.halign) {
982 case RS_MTextData::HACenter:
983 textLine->move(RS_Vector{-0.5 * (textLine->getMin().x + textLine->getMax().x), 0.});
984 break;
985
986 case RS_MTextData::HARight:
987 textLine->move(RS_Vector{-textLine->getMax().x, 0.});
988 break;
989
990 default:
991 textLine->move(RS_Vector{-textLine->getMin().x, 0.});
992 break;
993 }
994
995 // Update actual text size (before rotating, after scaling!):
996 if (textLine->getSize().x > m_usedTextWidth) {
997 m_usedTextWidth = textLine->getSize().x;
998 }
999
1000 m_usedTextHeight += m_data.height * m_data.lineSpacingFactor * ls;
1001
1002 // Gets the distance over text base-line (before rotating, after scaling!):
1003 const double textTail = textLine->getMin().y;
1004
1005 // Move:
